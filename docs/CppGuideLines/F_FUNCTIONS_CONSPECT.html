F: Функции - Подробный конспект

О разделе: Раздел F содержит рекомендации по написанию функций. Хорошая функция выполняет одну задачу, имеет понятное имя, правильно передаёт параметры и возвращает значения. Функции - это основа модульности кода.

F.1: "Упаковывайте" осмысленные операции в тщательно названные функции

Проблема/Контекст

Повторяющийся код сложно поддерживать. Если логика изменяется, нужно обновлять её во всех местах. Вынесение общей логики в функцию делает код более читаемым и упрощает изменения.

Решение

Выделяйте повторяющиеся операции в отдельные функции с понятными именами. Имя функции должно точно описывать, что она делает.

❌ Как НЕ надо

// Плохо: повторяющийся код
void process_users() {
    // ... код для пользователя 1 ...
    if (user1.age >= 18 && user1.verified) {
        user1.status = "active";
    }
    
    // ... код для пользователя 2 ...
    if (user2.age >= 18 && user2.verified) {
        user2.status = "active";
    }
    
    // ... код для пользователя 3 ...
    if (user3.age >= 18 && user3.verified) {
        user3.status = "active";
    }
}

// Плохо: неясное имя функции
void do_stuff(int x, int y) {
    // что делает эта функция?
}

// Плохо: функция делает слишком много
void process_order(Order& order) {
    // валидация
    if (order.items.empty()) return;
    if (order.total < 0) return;
    
    // вычисление скидки
    double discount = 0;
    if (order.total > 1000) discount = 0.1;
    else if (order.total > 500) discount = 0.05;
    
    // применение скидки
    order.total *= (1 - discount);
    
    // сохранение в базу данных
    database.save(order);
    
    // отправка уведомления
    email_service.send(order.customer_email, "Order processed");
}

✅ Как надо

// Хорошо: вынесенная функция
bool is_adult_and_verified(const User& user) {
    return user.age >= 18 && user.verified;
}

void activate_user(User& user) {
    if (is_adult_and_verified(user)) {
        user.status = "active";
    }
}

void process_users() {
    activate_user(user1);
    activate_user(user2);
    activate_user(user3);
}

// Хорошо: понятное имя функции
int calculate_distance(int x1, int y1, int x2, int y2) {
    int dx = x2 - x1;
    int dy = y2 - y1;
    return std::sqrt(dx * dx + dy * dy);
}

// Хорошо: разделение на логические функции
bool is_valid_order(const Order& order) {
    return !order.items.empty() && order.total >= 0;
}

double calculate_discount(double total) {
    if (total > 1000) return 0.1;
    if (total > 500) return 0.05;
    return 0.0;
}

void apply_discount(Order& order) {
    double discount = calculate_discount(order.total);
    order.total *= (1 - discount);
}

void process_order(Order& order) {
    if (!is_valid_order(order)) return;
    
    apply_discount(order);
    database.save(order);
    email_service.send(order.customer_email, "Order processed");
}

F.2: Функция должна выполнять одну логическую операцию

Проблема/Контекст

Функция, которая делает несколько вещей, сложна для понимания, тестирования и повторного использования. Если вам нужно использовать только часть функциональности, вы вынуждены использовать всю функцию.

Решение

Разделяйте функции так, чтобы каждая выполняла одну логическую операцию. Если функция делает несколько вещей, разделите её.

❌ Как НЕ надо

// Плохо: функция делает несколько вещей
void read_and_process_and_save(const std::string& filename) {
    // чтение файла
    std::ifstream file(filename);
    std::string content;
    std::string line;
    while (std::getline(file, line)) {
        content += line + "\n";
    }
    
    // обработка данных
    std::transform(content.begin(), content.end(), content.begin(), ::toupper);
    
    // сохранение
    std::ofstream out("output.txt");
    out << content;
}

// Плохо: функция выполняет несколько операций
void user_registration(const std::string& email, const std::string& password) {
    // валидация
    if (email.empty() || password.empty()) {
        throw std::invalid_argument("Email and password required");
    }
    
    // создание пользователя
    User user;
    user.email = email;
    user.password = hash_password(password);
    
    // сохранение в базу
    database.save(user);
    
    // отправка приветственного письма
    email_service.send_welcome(email);
    
    // логирование
    logger.log("User registered: " + email);
}

✅ Как надо

// Хорошо: каждая функция делает одну вещь
std::string read_file(const std::string& filename) {
    std::ifstream file(filename);
    std::string content;
    std::string line;
    while (std::getline(file, line)) {
        content += line + "\n";
    }
    return content;
}

std::string to_uppercase(const std::string& str) {
    std::string result = str;
    std::transform(result.begin(), result.end(), result.begin(), ::toupper);
    return result;
}

void save_to_file(const std::string& filename, const std::string& content) {
    std::ofstream out(filename);
    out << content;
}

// Использование:
void process_file(const std::string& filename) {
    auto content = read_file(filename);
    auto processed = to_uppercase(content);
    save_to_file("output.txt", processed);
}

// Хорошо: разделение на отдельные функции
bool is_valid_registration_data(const std::string& email, const std::string& password) {
    return !email.empty() && !password.empty();
}

User create_user(const std::string& email, const std::string& password) {
    User user;
    user.email = email;
    user.password = hash_password(password);
    return user;
}

void register_user(const User& user) {
    database.save(user);
    email_service.send_welcome(user.email);
    logger.log("User registered: " + user.email);
}

// Использование:
void user_registration(const std::string& email, const std::string& password) {
    if (!is_valid_registration_data(email, password)) {
        throw std::invalid_argument("Email and password required");
    }
    
    auto user = create_user(email, password);
    register_user(user);
}

F.3: Держите функции короткими и простыми

Проблема/Контекст

Длинные функции сложно читать, понимать и тестировать. В них легко допустить ошибки, и сложно найти, где именно проблема.

Решение

Держите функции короткими. Если функция не помещается на экран, она, вероятно, слишком длинная. Разбивайте длинные функции на более короткие.

Правило: "Не помещается на экран" часто является хорошим практическим определением "слишком большой". Функции от одной до пяти строк должны считаться нормальными.

❌ Как НЕ надо

// Плохо: очень длинная функция (100+ строк)
void process_data() {
    // инициализация переменных
    std::vector<int> data;
    std::vector<int> filtered;
    std::vector<int> sorted;
    int sum = 0;
    double average = 0.0;
    
    // чтение данных
    std::ifstream file("data.txt");
    std::string line;
    while (std::getline(file, line)) {
        int value = std::stoi(line);
        data.push_back(value);
    }
    
    // фильтрация
    for (int x : data) {
        if (x > 0 && x < 100) {
            filtered.push_back(x);
        }
    }
    
    // сортировка
    std::sort(filtered.begin(), filtered.end());
    
    // вычисление суммы
    for (int x : filtered) {
        sum += x;
    }
    
    // вычисление среднего
    if (!filtered.empty()) {
        average = static_cast<double>(sum) / filtered.size();
    }
    
    // вывод результатов
    std::cout << "Filtered data: ";
    for (int x : filtered) {
        std::cout << x << " ";
    }
    std::cout << "\nSum: " << sum << "\n";
    std::cout << "Average: " << average << "\n";
    
    // сохранение в файл
    std::ofstream out("result.txt");
    out << sum << "\n" << average << "\n";
    
    // ... ещё 50 строк кода ...
}

✅ Как надо

// Хорошо: короткие функции, каждая делает одну вещь
std::vector<int> read_data_from_file(const std::string& filename) {
    std::vector<int> data;
    std::ifstream file(filename);
    std::string line;
    while (std::getline(file, line)) {
        data.push_back(std::stoi(line));
    }
    return data;
}

std::vector<int> filter_data(const std::vector<int>& data) {
    std::vector<int> filtered;
    for (int x : data) {
        if (x > 0 && x < 100) {
            filtered.push_back(x);
        }
    }
    return filtered;
}

int calculate_sum(const std::vector<int>& data) {
    int sum = 0;
    for (int x : data) {
        sum += x;
    }
    return sum;
}

double calculate_average(const std::vector<int>& data) {
    if (data.empty()) return 0.0;
    return static_cast<double>(calculate_sum(data)) / data.size();
}

void print_results(const std::vector<int>& data, int sum, double average) {
    std::cout << "Filtered data: ";
    for (int x : data) {
        std::cout << x << " ";
    }
    std::cout << "\nSum: " << sum << "\n";
    std::cout << "Average: " << average << "\n";
}

void save_results(const std::string& filename, int sum, double average) {
    std::ofstream out(filename);
    out << sum << "\n" << average << "\n";
}

// Главная функция - короткая и понятная
void process_data() {
    auto data = read_data_from_file("data.txt");
    auto filtered = filter_data(data);
    auto sorted = filtered;
    std::sort(sorted.begin(), sorted.end());
    
    int sum = calculate_sum(sorted);
    double average = calculate_average(sorted);
    
    print_results(sorted, sum, average);
    save_results("result.txt", sum, average);
}

F.6: Если ваша функция не должна выбрасывать исключения, объявите её noexcept

Проблема/Контекст

Объявление функции noexcept даёт компилятору гарантию, что функция не выбросит исключение. Это позволяет компилятору применять более агрессивные оптимизации.

Решение

Если функция гарантированно не выбрасывает исключения, объявите её noexcept. Особенно важно для деструкторов, операций перемещения и функций, которые вызываются в критических местах.

❌ Как НЕ надо

// Плохо: noexcept не указан, хотя функция не выбрасывает исключений
class MyClass {
public:
    ~MyClass() {
        // простой деструктор, не выбрасывает исключений
        delete ptr;
    }
    
    int get_value() const {
        return value;  // не выбрасывает исключений
    }
    
    void swap(MyClass& other) {
        std::swap(value, other.value);
        std::swap(ptr, other.ptr);
    }
};

// Плохо: noexcept(false) для функции, которая не выбрасывает исключений
int add(int a, int b) noexcept(false) {
    return a + b;  // никогда не выбрасывает исключений
}

✅ Как надо

// Хорошо: noexcept указан явно
class MyClass {
public:
    ~MyClass() noexcept {
        delete ptr;  // delete не выбрасывает исключений
    }
    
    int get_value() const noexcept {
        return value;  // простой доступ не выбрасывает исключений
    }
    
    void swap(MyClass& other) noexcept {
        std::swap(value, other.value);
        std::swap(ptr, other.ptr);  // swap не выбрасывает исключений
    }
};

// Хорошо: noexcept для простых операций
int add(int a, int b) noexcept {
    return a + b;  // арифметика не выбрасывает исключений
}

// Хорошо: условный noexcept
template<typename T>
void my_swap(T& a, T& b) noexcept(std::is_nothrow_move_constructible_v<T>) {
    T temp = std::move(a);
    a = std::move(b);
    b = std::move(temp);
}
Важно: Если вы объявили функцию noexcept, но она всё же выбрасывает исключение, программа завершится вызовом std::terminate(). Поэтому объявляйте noexcept только для функций, которые гарантированно не выбрасывают исключений.

F.16: Для "входных" параметров передавайте дешёвые для копирования типы по значению, остальные — по ссылке на const

Проблема/Контекст

Передача параметров по значению создаёт копию объекта, что может быть дорого для больших объектов. Передача по ссылке на const избегает копирования, но добавляет уровень косвенности.

Решение

Для небольших типов (int, double, указатели) передавайте по значению. Для больших типов (string, vector, пользовательские классы) передавайте по ссылке на const.

Как это работает?

Передача по значению создаёт копию объекта в стеке вызова функции. Для небольших типов это дешевле, чем разыменование указателя. Для больших типов копирование дорого, поэтому используется ссылка на const.

❌ Как НЕ надо

// Плохо: передача больших объектов по значению
void process_string(std::string s) {  // создаётся копия!
    // ...
}

void process_vector(std::vector<int> v) {  // создаётся копия!
    // ...
}

// Плохо: передача маленьких типов по ссылке
void add(const int& a, const int& b) {  // лишнее разыменование
    return a + b;
}

void multiply(const double& x, const double& y) {  // лишнее разыменование
    return x * y;
}

✅ Как надо

// Хорошо: большие объекты по ссылке на const
void process_string(const std::string& s) {  // без копирования
    // ...
}

void process_vector(const std::vector<int>& v) {  // без копирования
    // ...
}

// Хорошо: маленькие типы по значению
int add(int a, int b) {  // эффективно
    return a + b;
}

double multiply(double x, double y) {  // эффективно
    return x * y;
}

// Хорошо: указатели по значению (они маленькие)
void process_data(int* data, size_t size) {  // указатель копируется дешево
    // ...
}

Дополнительные примеры

// Пример: Правила передачи параметров
class LargeObject {
    char data[10000];
    // ...
};

// Маленькие типы - по значению
void func1(int x, double y, bool flag) { }

// Большие типы - по ссылке на const
void func2(const std::string& s, const std::vector<int>& v) { }
void func3(const LargeObject& obj) { }

// Указатели - по значению (они маленькие)
void func4(int* ptr, const char* str) { }

// Для move-only типов используйте rvalue ссылки
void func5(std::unique_ptr<int> ptr) { }  // принимает владение
void func6(std::unique_ptr<int>&& ptr) { }  // тоже принимает владение

F.17: Для "входно-выходных" параметров передавайте по ссылке на неконстанту

Проблема/Контекст

Когда функция должна изменить переданный объект, это должно быть явно видно в сигнатуре функции. Передача по ссылке на неконстанту делает это очевидным.

❌ Как НЕ надо

// Плохо: неясно, что функция изменяет параметр
void update(int value) {  // изменяет ли value?
    // ...
}

// Плохо: использование указателя для входно-выходного параметра
void process(int* data) {  // data может быть nullptr
    if (data) {
        *data = 42;
    }
}

// Использование:
int x = 10;
process(&x);  // нужно передавать адрес
if (x == 42) {  // но неясно, изменился ли x
    // ...
}

✅ Как надо

// Хорошо: ссылка на неконстанту явно показывает изменение
void update(int& value) {  // явно видно, что value изменяется
    value = 42;
}

// Использование:
int x = 10;
update(x);  // просто передаём переменную
// x теперь 42

// Хорошо: ссылка на неконстанту для входно-выходного параметра
void process(int& data) {  // data не может быть nullptr
    data = 42;  // безопасно изменяем
}

// Использование:
int x = 10;
process(x);  // x изменён на 42

F.20: Для "выходных" значений предпочитайте возвращаемые значения выходным параметрам

Проблема/Контекст

Использование выходных параметров делает код менее читаемым. Неясно, какие параметры являются входными, а какие выходными. Возвращаемые значения самодокументируемы.

❌ Как НЕ надо

// Плохо: выходной параметр вместо возвращаемого значения
void compute_result(int input, int& result) {  // result - выходной параметр
    result = input * 2;
}

// Использование:
int value;
compute_result(5, value);  // неочевидно, что value будет изменён
// value теперь 10

// Плохо: несколько выходных параметров
void find_min_max(const std::vector<int>& data, int& min, int& max) {
    if (data.empty()) {
        // что делать? min и max не инициализированы
        return;
    }
    min = *std::min_element(data.begin(), data.end());
    max = *std::max_element(data.begin(), data.end());
}

// Использование:
int min_val, max_val;  // неинициализированные переменные
find_min_max(vec, min_val, max_val);  // min_val и max_val изменены

✅ Как надо

// Хорошо: возвращаемое значение вместо выходного параметра
int compute_result(int input) {  // явно возвращает результат
    return input * 2;
}

// Использование:
int value = compute_result(5);  // очевидно, что value - результат
// value = 10

// Хорошо: структура для нескольких возвращаемых значений
struct MinMax {
    int min;
    int max;
};

MinMax find_min_max(const std::vector<int>& data) {
    if (data.empty()) {
        throw std::invalid_argument("Empty vector");
    }
    return {
        *std::min_element(data.begin(), data.end()),
        *std::max_element(data.begin(), data.end())
    };
}

// Использование:
auto result = find_min_max(vec);  // явно получаем результат
// result.min и result.max содержат значения

// Альтернатива: std::pair или std::tuple
std::pair<int, int> find_min_max_pair(const std::vector<int>& data) {
    if (data.empty()) {
        throw std::invalid_argument("Empty vector");
    }
    return {
        *std::min_element(data.begin(), data.end()),
        *std::max_element(data.begin(), data.end())
    };
}

// Использование:
auto [min_val, max_val] = find_min_max_pair(vec);  // structured binding
Исключение: Иногда выходные параметры оправданы, когда нужно вернуть несколько значений и создание структуры избыточно, или когда нужно избежать копирования больших объектов. Но в большинстве случаев предпочитайте возвращаемые значения.

Заключение

Ключевые принципы раздела F (Функции):

  • Одна задача: Функция должна выполнять одну логическую операцию
  • Короткие функции: Функция должна помещаться на экран
  • Понятные имена: Имя функции должно точно описывать её назначение
  • noexcept: Объявляйте noexcept для функций, которые не выбрасывают исключений
  • Правильная передача параметров: Маленькие типы по значению, большие по ссылке на const
  • Возвращаемые значения: Предпочитайте возвращаемые значения выходным параметрам
Правило: Хорошая функция - это функция, которую легко понять, протестировать и использовать. Если функция слишком сложная, разделите её на более простые функции.
