E: Обработка ошибок - Подробный конспект

О разделе: Раздел E содержит рекомендации по обработке ошибок. Правильная обработка ошибок делает программу надёжной и предсказуемой. Исключения - основной механизм обработки ошибок в C++.

E.2: Выбросьте исключение, чтобы сигнализировать, что функция не может выполнить свою задачу

❌ Как НЕ надо

// Плохо: возврат специального значения
int divide(int a, int b) {
    if (b == 0) {
        return -1;  // что означает -1? Ошибка?
    }
    return a / b;
}

int result = divide(10, 0);
if (result == -1) {  // легко забыть проверить
    // обработка ошибки
}
result = result * 2;  // используем -1 как обычное значение!

✅ Как надо

// Хорошо: исключение при ошибке
#include <stdexcept>

int divide(int a, int b) {
    if (b == 0) {
        throw std::invalid_argument("Деление на ноль");
    }
    return a / b;
}

try {
    int result = divide(10, 0);
    // этот код не выполнится, если b == 0
} catch (const std::invalid_argument& e) {
    // ошибка обработана явно
    std::cerr << "Ошибка: " << e.what() << "\n";
}

E.3: Используйте исключения только для обработки ошибок

❌ Как НЕ надо

// Плохо: исключение для нормального потока выполнения
int find_index(const std::vector<std::string>& vec, const std::string& x) {
    try {
        for (size_t i = 0; i < vec.size(); ++i) {
            if (vec[i] == x) throw i;  // исключение для нормального случая!
        }
    } catch (size_t i) {
        return i;
    }
    return -1;
}

✅ Как надо

// Хорошо: стандартный алгоритм для нормального потока
#include <algorithm>

int find_index(const std::vector<std::string>& vec, const std::string& x) {
    auto it = std::find(vec.begin(), vec.end(), x);
    if (it != vec.end()) {
        return std::distance(vec.begin(), it);
    }
    return -1;  // нормальное значение для "не найдено"
}

// Или используйте std::optional для опциональных значений
#include <optional>
std::optional<size_t> find_index_opt(const std::vector<std::string>& vec, 
                                        const std::string& x) {
    auto it = std::find(vec.begin(), vec.end(), x);
    if (it != vec.end()) {
        return std::distance(vec.begin(), it);
    }
    return std::nullopt;  // отсутствие значения - не ошибка
}

E.14: Используйте специально разработанные пользовательские типы как исключения

❌ Как НЕ надо

// Плохо: встроенные типы как исключения
throw 7;  // что означает 7?
throw "something bad";  // строка - плохо
throw std::exception{};  // пустое исключение

✅ Как надо

// Хорошо: пользовательские типы исключений
#include <stdexcept>
#include <string>

class FileNotFoundError : public std::runtime_error {
public:
    explicit FileNotFoundError(const std::string& filename)
        : std::runtime_error("Файл не найден: " + filename) {}
};

class DivisionByZeroError : public std::runtime_error {
public:
    DivisionByZeroError()
        : std::runtime_error("Деление на ноль") {}
};

// Использование:
throw FileNotFoundError("config.txt");
throw DivisionByZeroError();

E.15: Выбрасывайте по значению, ловите исключения из иерархии по ссылке

❌ Как НЕ надо

// Плохо: выбрасывание по указателю
void f() {
    try {
        throw new MyException("error");  // утечка памяти!
    } catch (MyException* e) {
        // кто удалит e?
        delete e;  // легко забыть
    }
}

// Плохо: ловля по значению (срезание)
void g() {
    try {
        throw DerivedException();
    } catch (BaseException e) {  // срезание! теряется информация
        // ...
    }
}

✅ Как надо

// Хорошо: выбрасывание по значению, ловля по ссылке
void f() {
    try {
        throw MyException("error");  // по значению
    } catch (const MyException& e) {  // по ссылке на const
        // автоматическое управление памятью
        std::cerr << e.what() << "\n";
    }
}

// Хорошо: ловля по ссылке (без срезания)
void g() {
    try {
        throw DerivedException();
    } catch (const BaseException& e) {  // по ссылке, без срезания
        // можно использовать dynamic_cast, если нужно
        // ...
    }
}

E.6: Используйте RAII для предотвращения утечек

RAII: Ресурс приобретается в конструкторе и освобождается в деструкторе. Это гарантирует освобождение ресурсов даже при исключениях.

❌ Как НЕ надо

// Плохо: ручное управление ресурсами
void process_file(const char* filename) {
    FILE* file = fopen(filename, "r");
    if (!file) {
        throw std::runtime_error("Cannot open file");
    }
    
    // ... обработка ...
    if (error) {
        fclose(file);  // нужно не забыть
        throw std::runtime_error("Processing error");
    }
    
    fclose(file);  // легко забыть
}

✅ Как надо

// Хорошо: RAII автоматически освобождает ресурсы
#include <fstream>

void process_file(const std::string& filename) {
    std::ifstream file(filename);
    if (!file.is_open()) {
        throw std::runtime_error("Cannot open file");
    }
    
    // ... обработка ...
    if (error) {
        throw std::runtime_error("Processing error");
        // файл автоматически закроется
    }
    // файл автоматически закроется при выходе
}

Заключение

Ключевые принципы раздела E (Обработка ошибок):

  • Исключения для ошибок: Используйте исключения для сигнализации об ошибках
  • Только для ошибок: Не используйте исключения для нормального потока выполнения
  • Пользовательские типы: Создавайте специализированные типы исключений
  • Выбрасывание по значению: Выбрасывайте исключения по значению
  • Ловля по ссылке: Ловите исключения по ссылке на const
  • RAII: Используйте RAII для автоматического управления ресурсами
