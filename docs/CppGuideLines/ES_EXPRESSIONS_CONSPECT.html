ES: Выражения и операторы - Подробный конспект

О разделе: Раздел ES содержит рекомендации по написанию выражений и операторов. Правильное использование конструкций языка делает код более читаемым, безопасным и эффективным.

ES.5: Держите области видимости маленькими

❌ Как НЕ надо

void process() {
    int i;  // область видимости слишком большая
    for (i = 0; i < 10; ++i) {
        // ...
    }
    // i всё ещё доступна здесь, хотя не нужна
    int j = i * 2;  // может использовать старое значение
}

✅ Как надо

void process() {
    for (int i = 0; i < 10; ++i) {
        // i доступна только в цикле
        // ...
    }
    // i недоступна здесь
}

ES.20: Всегда инициализируйте объект

❌ Как НЕ надо

void use() {
    int i;  // неинициализирована!
    // ...
    i = 7;  // использование до инициализации - ОПАСНО
    
    int* ptr;  // неинициализирован!
    *ptr = 42;  // неопределённое поведение!
}

✅ Как надо

void use() {
    int i = 7;  // инициализирована сразу
    // ...
    
    int* ptr = nullptr;  // инициализирован
    if (ptr) {
        *ptr = 42;  // безопасно
    }
}

ES.23: Предпочитайте синтаксис инициализации {}

❌ Как НЕ надо

int x(7.9);  // сужающее преобразование - не ошибка!
int y = 7.9;  // тоже сужающее преобразование

// Небезопасно: потеря данных без предупреждения

✅ Как надо

int x{7.9};  // ошибка компиляции: сужающее преобразование
int y{7};    // ОК: точное совпадение типов

// Безопасно: компилятор предотвращает потерю данных

ES.47: Используйте nullptr, а не 0 или NULL

❌ Как НЕ надо

void f(int);
void f(char*);

f(0);     // вызов f(int) - не то, что нужно!
f(NULL);  // может быть f(int) или f(char*) - неопределённо!

✅ Как надо

void f(int);
void f(char*);

f(nullptr);  // явно вызов f(char*)

int* ptr = nullptr;  // явно указатель
if (ptr == nullptr) {  // понятно
    // ...
}

ES.48: Избегайте приведений типов

❌ Как НЕ надо

double d = 3.14;
int i = (int)d;  // C-style cast - опасно!

void* ptr = &d;
double* dp = (double*)ptr;  // небезопасно!

Base* b = new Derived();
Derived* d = (Derived*)b;  // небезопасное приведение!

✅ Как надо

double d = 3.14;
int i = static_cast<int>(d);  // явное преобразование

void* ptr = &d;
double* dp = static_cast<double*>(ptr);  // безопасно

Base* b = new Derived();
Derived* d = dynamic_cast<Derived*>(b);  // безопасное приведение
if (d) {
    // использование d
}

ES.56: Пишите std::move() только когда нужно явно переместить

❌ Как НЕ надо

std::vector<int> make_vector() {
    std::vector<int> result;
    // ... заполнение ...
    return std::move(result);  // ненужный move!
    // Компилятор сам оптимизирует через RVO
}

✅ Как надо

std::vector<int> make_vector() {
    std::vector<int> result;
    // ... заполнение ...
    return result;  // компилятор сам переместит или использует RVO
}

// move нужен только для явной передачи владения:
void take_ownership(std::unique_ptr<int> ptr) { }
auto ptr = std::make_unique<int>(42);
take_ownership(std::move(ptr));  // явная передача владения

ES.70: Предпочитайте switch оператор if-оператору

❌ Как НЕ надо

void process(int n) {
    if (n == 0) {
        // ...
    } else if (n == 1) {
        // ...
    } else if (n == 2) {
        // ...
    } else {
        // ...
    }
}

✅ Как надо

void process(int n) {
    switch (n) {
    case 0:
        // ...
        break;
    case 1:
        // ...
        break;
    case 2:
        // ...
        break;
    default:
        // ...
        break;
    }
}

ES.71: Предпочитайте range-for оператор for-оператору

❌ Как НЕ надо

std::vector<int> vec = {1, 2, 3, 4, 5};

for (size_t i = 0; i < vec.size(); ++i) {
    std::cout << vec[i] << " ";  // возможна ошибка индекса
}

// Ошибка: выход за границы
for (size_t i = 0; i <= vec.size(); ++i) {
    std::cout << vec[i] << " ";  // vec[vec.size()] - ошибка!
}

✅ Как надо

std::vector<int> vec = {1, 2, 3, 4, 5};

for (const auto& x : vec) {
    std::cout << x << " ";  // безопасно, невозможно выйти за границы
}

// Для изменения элементов:
for (auto& x : vec) {
    x *= 2;  // безопасное изменение
}

Заключение

Ключевые принципы раздела ES (Выражения и операторы):

  • Маленькие области видимости: Ограничивайте область видимости переменных
  • Инициализация: Всегда инициализируйте переменные
  • {} инициализация: Используйте фигурные скобки для безопасной инициализации
  • nullptr: Используйте nullptr вместо 0 или NULL
  • Избегайте приведений: Используйте named casts вместо C-style casts
  • move только когда нужно: Не используйте move без необходимости
  • switch вместо if: Используйте switch для множественного выбора
  • range-for: Предпочитайте range-for обычным циклам
