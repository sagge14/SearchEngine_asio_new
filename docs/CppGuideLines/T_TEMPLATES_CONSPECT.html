T: Шаблоны и обобщённое программирование - Подробный конспект

О разделе: Раздел T содержит рекомендации по использованию шаблонов и обобщённого программирования. Шаблоны позволяют писать код, работающий с различными типами, повышая уровень абстракции и переиспользование кода.

T.1: Используйте шаблоны для повышения уровня абстракции кода

❌ Как НЕ надо

// Плохо: дублирование кода для разных типов
int find_int(const std::vector<int>& vec, int value) {
    for (const auto& x : vec) {
        if (x == value) return &x - &vec[0];
    }
    return -1;
}

double find_double(const std::vector<double>& vec, double value) {
    for (const auto& x : vec) {
        if (x == value) return &x - &vec[0];
    }
    return -1;
}
// Дублирование кода!

✅ Как надо

// Хорошо: шаблон для обобщения
#include <algorithm>

template<typename T>
auto find_index(const std::vector<T>& vec, const T& value) {
    auto it = std::find(vec.begin(), vec.end(), value);
    if (it != vec.end()) {
        return std::distance(vec.begin(), it);
    }
    return -1;
}

// Использование с любым типом:
std::vector<int> int_vec = {1, 2, 3};
int idx1 = find_index(int_vec, 2);

std::vector<double> double_vec = {1.1, 2.2, 3.3};
int idx2 = find_index(double_vec, 2.2);

T.10: Указывайте концепты для всех аргументов шаблона

❌ Как НЕ надо

// Плохо: нет ограничений на тип
template<typename T>
void process(T value) {
    value.doSomething();  // что если T не имеет doSomething()?
    // Ошибка будет только при инстанцировании
}

✅ Как надо

// Хорошо: концепты ограничивают типы (C++20)
#include <concepts>

template<std::integral T>
void process_integer(T value) {
    // T гарантированно целочисленный тип
}

// Или используйте requires:
template<typename T>
requires std::integral<T>
void process_integer(T value) {
    // T гарантированно целочисленный тип
}

// Или используйте стандартные концепты:
#include <iterator>
template<std::input_iterator Iter>
void process_range(Iter begin, Iter end) {
    // Iter гарантированно input iterator
}

T.11: Везде, где возможно, используйте стандартные концепты

Примечание: Стандартная библиотека C++20 предоставляет множество готовых концептов. Используйте их вместо создания собственных, когда это возможно.

❌ Как НЕ надо

// Плохо: создание собственного концепта, когда есть стандартный
template<typename T>
concept MyNumber = std::is_arithmetic_v<T>;

template<MyNumber T>
T add(T a, T b) {
    return a + b;
}

✅ Как надо

// Хорошо: использование стандартного концепта
#include <concepts>

template<std::arithmetic T>
T add(T a, T b) {
    return a + b;
}

// Стандартные концепты включают:
// - std::integral, std::floating_point, std::arithmetic
// - std::same_as, std::derived_from
// - std::convertible_to, std::common_reference_with
// - std::copyable, std::movable, std::semiregular
// - std::equality_comparable, std::totally_ordered
// - std::regular, std::invocable
// И многие другие...

T.40: Используйте функциональные объекты для передачи операций алгоритмам

❌ Как НЕ надо

// Плохо: указатель на функцию
bool greater(double x, double y) {
    return x > y;
}

std::vector<double> vec = {3.0, 1.0, 4.0, 2.0};
std::sort(vec.begin(), vec.end(), greater);  // указатель на функцию

✅ Как надо

// Хорошо: функциональный объект или лямбда
#include <algorithm>
#include <functional>

std::vector<double> vec = {3.0, 1.0, 4.0, 2.0};

// Вариант 1: стандартный функциональный объект
std::sort(vec.begin(), vec.end(), std::greater<>{});

// Вариант 2: лямбда
std::sort(vec.begin(), vec.end(), 
          [](double x, double y) { return x > y; });

// Вариант 3: пользовательский функциональный объект
struct Greater {
    bool operator()(double x, double y) const {
        return x > y;
    }
};
std::sort(vec.begin(), vec.end(), Greater{});

Заключение

Ключевые принципы раздела T (Шаблоны):

  • Повышение абстракции: Используйте шаблоны для обобщения кода
  • Концепты: Указывайте концепты для всех аргументов шаблона
  • Стандартные концепты: Используйте стандартные концепты, когда возможно
  • Функциональные объекты: Используйте функциональные объекты вместо указателей на функции
  • Простота: Не усложняйте шаблоны без необходимости
